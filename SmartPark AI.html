// Dynamic Surge Pricing Tier Engine
export function calculateDynamicSurgeRate(baseHourlyRate, occupancyRate, isPeakWindow, requiresEV) {
    let multiplier = 1.0;
    
    // Tiered Occupancy Multiplier
    if (occupancyRate >= 0.90) {
        multiplier = 2.0;       // Extreme Congestion
    } else if (occupancyRate >= 0.75) {
        multiplier = 1.5;       // High Surge
    } else if (occupancyRate >= 0.50) {
        multiplier = 1.25;      // Moderate Surge
    }

    // Rush Hour Addition
    if (isPeakWindow) {
        multiplier *= 1.15;
    }

    // EV High-Speed Charging Premium
    const evAddon = requiresEV ? 3.50 : 0.00;
    const finalRate = (baseHourlyRate * multiplier) + evAddon;

    return {
        baseRate: baseHourlyRate,
        multiplier: Number(multiplier.toFixed(2)),
        surgeTier: occupancyRate >= 0.90 ? 'Critical' : occupancyRate >= 0.75 ? 'High' : 'Normal',
        calculatedHourlyRate: Number(finalRate.toFixed(2))
    };
}
// Explainable AI Parking Deck Recommendation Engine
export function rankParkingLocations(locations, userLat, userLng, userBudget, needsEV) {
    return locations.map(loc => {
        const distanceKm = calculateHaversine(userLat, userLng, loc.latitude, loc.longitude);
        const vacancyRatio = loc.availableSlots / loc.totalSlots;
        
        // Multi-Criteria Weighted Utility Scoring
        const distScore = Math.max(0, 10 - distanceKm * 2);
        const vacancyScore = vacancyRatio * 10;
        const budgetScore = (userBudget >= loc.hourlyRate) ? 10 : Math.max(0, 10 - (loc.hourlyRate - userBudget));
        const evScore = (needsEV && loc.hasEvCharging) ? 10 : (needsEV ? 0 : 5);

        const totalScore = (distScore * 0.35) + (vacancyScore * 0.25) + (budgetScore * 0.20) + (evScore * 0.20);
        
        // Explainability Reasoning Tag Generation
        const reasons = [];
        if (distanceKm < 1.0) reasons.push('🚶 Walking Distance (<1km)');
        if (loc.hasEvCharging && needsEV) reasons.push('⚡ Fast EV Charging Ready');
        if (vacancyRatio > 0.4) reasons.push('🟢 High Spot Availability');
        if (loc.hourlyRate <= userBudget) reasons.push('💰 Under Target Budget');

        return {
            ...loc,
            aiMatchScore: Math.round(totalScore * 10),
            explainablePills: reasons
        };
    }).sort((a, b) => b.aiMatchScore - a.aiMatchScore);
}
// Role-Based Access Control (RBAC) Guard Middleware
export function requireRole(allowedRoles = []) {
    return (req, res, next) => {
        const authHeader = req.headers['authorization'];
        if (!authHeader || !authHeader.startsWith('Bearer ')) {
            return res.status(401).json({ success: false, message: 'Missing or malformed JWT token.' });
        }

        const token = authHeader.split(' ')[1];
        try {
            const decoded = jwt.verify(token, process.env.JWT_SECRET);
            req.user = decoded;

            if (allowedRoles.length > 0 && !allowedRoles.includes(decoded.role)) {
                return res.status(403).json({ 
                    success: false, 
                    message: `Forbidden: Requires one of [${allowedRoles.join(', ')}] role.` 
                });
            }
            next();
        } catch (err) {
            return res.status(401).json({ success: false, message: 'Invalid or expired session token.' });
        }
    };
}
